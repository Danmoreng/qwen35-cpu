cmake_minimum_required(VERSION 3.24)
project(qwen35_cuda_calibration LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(LLAMA_SOURCE_DIR "" CACHE PATH "External pinned llama.cpp checkout")
if(NOT EXISTS "${LLAMA_SOURCE_DIR}/include/llama.h")
  message(FATAL_ERROR "Set LLAMA_SOURCE_DIR to the pinned llama.cpp checkout")
endif()
# Offline teacher only. This target is independent of the CPU inference engine
# and of the CPU comparison harness, which continues to force CUDA off.
set(GGML_CUDA ON CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE_DIR}" llama EXCLUDE_FROM_ALL)
add_executable(llama-calibration-cuda ../llama-comparison/main.cpp)
target_compile_definitions(llama-calibration-cuda PRIVATE QWEN35_CALIBRATION_CUDA=1)
target_link_libraries(llama-calibration-cuda PRIVATE llama ggml)
add_executable(llama-calibration-resident-cuda resident.cpp)
target_compile_definitions(llama-calibration-resident-cuda PRIVATE QWEN35_CALIBRATION_CUDA=1)
target_link_libraries(llama-calibration-resident-cuda PRIVATE llama ggml)
