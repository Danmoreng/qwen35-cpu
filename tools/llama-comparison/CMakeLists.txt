cmake_minimum_required(VERSION 3.24)
project(qwen35_cpu_llama_comparison LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(LLAMA_SOURCE_DIR "" CACHE PATH "External checkout of pinned llama.cpp")
if (NOT EXISTS "${LLAMA_SOURCE_DIR}/include/llama.h")
  message(FATAL_ERROR "Set LLAMA_SOURCE_DIR to a llama.cpp checkout")
endif ()
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE ON CACHE BOOL "" FORCE)
if (MSVC)
  # Upstream FindSIMD detects base AVX-512 on MSVC but does not enable VNNI.
  # Probe the host instead of silently comparing against a weaker dot kernel.
  include(CheckCXXSourceRuns)
  check_cxx_source_runs("#include <intrin.h>\nint main(){int r[4];__cpuidex(r,7,0);return (r[2]&(1<<11))?0:1;}" HOST_AVX512_VNNI)
  check_cxx_source_runs("#include <intrin.h>\nint main(){int r[4];__cpuidex(r,7,1);return (r[0]&(1<<4))?0:1;}" HOST_AVX_VNNI)
  if (HOST_AVX512_VNNI)
    set(GGML_AVX512_VNNI ON CACHE BOOL "" FORCE)
  endif ()
  if (HOST_AVX_VNNI)
    set(GGML_AVX_VNNI ON CACHE BOOL "" FORCE)
  endif ()
endif ()
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE_DIR}" llama EXCLUDE_FROM_ALL)
add_executable(llama-fixed-cpu-bench main.cpp)
target_link_libraries(llama-fixed-cpu-bench PRIVATE llama ggml)
