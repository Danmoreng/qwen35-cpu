cmake_minimum_required(VERSION 3.24)
project(qwen35_cpu_ik_llama_comparison LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(IK_LLAMA_SOURCE_DIR "" CACHE PATH "External pinned ik_llama.cpp checkout")
if (NOT EXISTS "${IK_LLAMA_SOURCE_DIR}/include/llama.h")
  message(FATAL_ERROR "Set IK_LLAMA_SOURCE_DIR to an ik_llama.cpp checkout")
endif ()
foreach(backend CUDA VULKAN METAL HIPBLAS SYCL MUSA RPC)
  set(GGML_${backend} OFF CACHE BOOL "" FORCE)
endforeach()
set(GGML_NATIVE ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory("${IK_LLAMA_SOURCE_DIR}" ik EXCLUDE_FROM_ALL)
add_executable(ik-fixed-cpu-bench ../llama-comparison/main.cpp)
target_compile_definitions(ik-fixed-cpu-bench PRIVATE QWEN35_IK_LLAMA=1)
target_link_libraries(ik-fixed-cpu-bench PRIVATE llama ggml)
add_subdirectory("${IK_LLAMA_SOURCE_DIR}/examples/quantize" ik-quantize EXCLUDE_FROM_ALL)
